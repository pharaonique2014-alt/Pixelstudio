cmake_minimum_required(VERSION 3.18)
project(jarvis_sd CXX C)
set(CMAKE_CXX_STANDARD 17)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(${SDCPP_DIR} sdcpp)
add_library(jarvis_sd SHARED jarvis_sd.cpp)
target_include_directories(jarvis_sd PRIVATE ${SDCPP_DIR}/include)
target_link_libraries(jarvis_sd PRIVATE stable-diffusion log)
target_link_options(jarvis_sd PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
